export function parseRatingChanges(result: string, ratings: number[], kFactor: number): number[] {
  const scores = result
    .split(/[:-]/)
    .map(Number)
    .filter((score) => !Number.isNaN(score));

  const deltas = ratings.map(() => 0);

  for (let i = 0; i < ratings.length; i++) {
    for (let j = i + 1; j < ratings.length; j++) {
      const expectedI = 1 / (1 + Math.pow(10, (ratings[j] - ratings[i]) / 400));
      const expectedJ = 1 / (1 + Math.pow(10, (ratings[i] - ratings[j]) / 400));

      let actualI = 0.5;
      let actualJ = 0.5;

      if (scores[i] > scores[j]) {
        actualI = 1;
        actualJ = 0;
      } else if (scores[i] < scores[j]) {
        actualI = 0;
        actualJ = 1;
      }

      deltas[i] += kFactor * (actualI - expectedI);
      deltas[j] += kFactor * (actualJ - expectedJ);
    }
  }

  return deltas.map((delta) => Math.round(delta));
}
